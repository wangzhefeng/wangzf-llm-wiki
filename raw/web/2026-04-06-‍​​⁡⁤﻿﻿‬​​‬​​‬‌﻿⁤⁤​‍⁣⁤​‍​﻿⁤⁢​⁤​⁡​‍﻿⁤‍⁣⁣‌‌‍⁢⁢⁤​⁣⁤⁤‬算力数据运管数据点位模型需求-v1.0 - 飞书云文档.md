Job 级别业务数据需求

训练任务运管数据算法需要的指标：

select

uid,

state,

status::json -> 'statusPhase' as status\_phase,

status::json -> 'exceptionReason' ->> 'lastUpdateTime' as last\_updateTime,

status::json -> 'lifecycle' ->> 'startTimestamp' as start\_timestamp,

status::json -> 'lifecycle' ->> 'completionTimestamp' as completion\_timestamp,

status::json -> 'lifecycle' ->> 'runningDuration' as running\_duration,

status::json -> 'pods' as pods,

\-- status::json -> jsonb\_array\_elements('podDetails'::jsonb) -> 'name' as pod\_name,

\-- status::json -> jsonb\_array\_elements('podDetails'::jsonb) -> 'containers' as pod\_containers,

jsonb\_array\_elements(task\_roles::jsonb) -> 'roleName' as role\_name,

jsonb\_array\_elements(task\_roles::jsonb) -> 'instances' as instances,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'spec' as resources\_spec,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'limits' -> 'cpu' as resources\_limits\_cpu,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'limits' -> 'memory' as resources\_limits\_memory,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'limits' -> 'nvidia.com/gpu' as resources\_limits\_gpu,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'limits' -> 'rdma/hca' as resources\_limits\_rdma\_hca,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'requests' -> 'cpu' as resources\_requests\_cpu,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'requests' -> 'memory' as resources\_requests\_memory,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'requests' -> 'nvidia.com/gpu' as resources\_requests\_gpu,

jsonb\_array\_elements(task\_roles::jsonb) -> 'resources' -> 'resources' -> 'requests' -> 'rdma/hca' as resources\_requests\_rdma\_hca,

training\_type

from "public"."lepton\_service\_training\_jobs"

where updated\_at >= '2026-03-01' and updated\_at <= '2026-04-01'

\-- limit 10
